cmake_minimum_required(VERSION 3.28)
project(PlussBI LANGUAGES CXX)
include(GNUInstallDirs)

# Embed the real frontend sources (frontend/static/css/dashboard.css,
# frontend/js/dashboard.js, frontend/html/dashboard_shell.html) into a
# generated C++ header at configure time, byte-for-byte, so they stay real
# editable .css/.js/.html files instead of C++ string literals baked into
# studio.h. CMAKE_CONFIGURE_DEPENDS makes CMake reconfigure (regenerating
# the header) whenever one of the source files changes.
#
# This is written directly into the source tree, under
# include/pluss/bi/ next to studio.h itself, rather than into a
# build-tree "generated/" include directory the way PlussBI's own PUBLIC
# include dirs normally would: several sibling modules (PlussAI, PlussMa,
# and their submodules) add a hardcoded "../PlussTils/include" path to the
# top-level `pluss` umbrella target directly in their own CMakeLists.txt,
# bypassing PlussTils'/PlussBI's target-level usage requirements entirely.
# A build-tree path would be invisible to those; include/pluss/bi/ is
# already on every consumer's path unconditionally. The generated file is
# gitignored -- see the repo's top-level .gitignore.
set(PLUSSBI_FRONTEND_DIR ${CMAKE_CURRENT_SOURCE_DIR}/frontend)
set(PLUSSBI_CSS_FILE ${PLUSSBI_FRONTEND_DIR}/static/css/dashboard.css)
set(PLUSSBI_JS_FILE ${PLUSSBI_FRONTEND_DIR}/js/dashboard.js)
set(PLUSSBI_HTML_FILE ${PLUSSBI_FRONTEND_DIR}/html/dashboard_shell.html)

file(READ ${PLUSSBI_CSS_FILE} PLUSSBI_CSS_CONTENT)
file(READ ${PLUSSBI_JS_FILE} PLUSSBI_JS_CONTENT)
file(READ ${PLUSSBI_HTML_FILE} PLUSSBI_HTML_CONTENT)

set(PLUSSBI_ASSETS_HEADER ${CMAKE_CURRENT_SOURCE_DIR}/../../include/pluss/bi/dashboard_assets.h)

file(WRITE ${PLUSSBI_ASSETS_HEADER}
"#pragma once\n"
"\n"
"// GENERATED by src/PlussBI/CMakeLists.txt from the real .css/.js/.html\n"
"// sources under src/PlussBI/frontend/. Edit those files, not this one --\n"
"// this header is regenerated automatically the next time CMake\n"
"// reconfigures (it watches the source files via CMAKE_CONFIGURE_DEPENDS).\n"
"\n"
"namespace pluss::bi::assets {\n"
"\n"
"inline constexpr const char* kDashboardCss = R\"PLUSSBI_ASSET(${PLUSSBI_CSS_CONTENT})PLUSSBI_ASSET\";\n"
"\n"
"inline constexpr const char* kDashboardJs = R\"PLUSSBI_ASSET(${PLUSSBI_JS_CONTENT})PLUSSBI_ASSET\";\n"
"\n"
"inline constexpr const char* kDashboardShellHtml = R\"PLUSSBI_ASSET(${PLUSSBI_HTML_CONTENT})PLUSSBI_ASSET\";\n"
"\n"
"} // namespace pluss::bi::assets\n"
)

set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${PLUSSBI_CSS_FILE} ${PLUSSBI_JS_FILE} ${PLUSSBI_HTML_FILE})

add_library(PlussBI STATIC
    ${CMAKE_CURRENT_SOURCE_DIR}/runtime/BrowserExport.cpp
)
add_library(Pluss::BI ALIAS PlussBI)

target_include_directories(PlussBI PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/../../include>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/httplib>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

find_package(Threads REQUIRED)
target_link_libraries(PlussBI PUBLIC Threads::Threads)

set_target_properties(PlussBI PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF
)

install(TARGETS PlussBI
    EXPORT PlussTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

# dashboard_assets.h is generated straight into include/pluss/bi/ above, so
# it is already covered by whatever installs the rest of that directory --
# no separate install() rule needed here.
