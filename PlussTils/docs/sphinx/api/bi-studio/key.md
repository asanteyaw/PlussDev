# `pluss::bi::Studio::key`

[Back to `Studio`](../bi-studio.md) · [PlussBI API Reference](../bi.md)

```cpp
#include <pluss/bi/specs.h>

const std::string& key() const noexcept;
```

`key` returns the studio's key, exactly as it was given. A default-constructed studio is keyed `studio`. The key names the studio and is used as the
page heading when a board or page has no title of its own. It does not change what the boards render.

## Parameters

Not applicable. The method takes no parameters.

## Returns

A reference to the key string.

## Errors

This call does not throw.

## Practical Examples

### Example 1

Example 1 shows that a key with spaces and digits is stored unchanged.

```cpp
#include <pluss/bi/bi.h>
#include <pluss/bi/serve.h>
#include <pluss/bundle/bundle.h>
#include <pluss/mic.h>

#include <cstdint>
#include <filesystem>
#include <fstream>
#include <iostream>
#include <iterator>
#include <string>
#include <vector>

namespace bi = pluss::bi;
namespace fs = std::filesystem;

// Three sales rows: regions north, south, north; amounts 10, 20, 30.
pluss::bundle::Nexus make_nexus() {
  using pluss::table::DataFrame;
  using pluss::table::Slate;
  const auto cfg = pluss::Config{}.with_dtype(pluss::ScalarType::Float64).with_platform(pluss::pCPU);
  const DataFrame sales({
      {"region", Slate(pluss::make_note({"north", "south", "north"}))},
      {"amount", Slate(pluss::make_series(std::vector<double>{10, 20, 30}, {3}, cfg))},
  });
  return pluss::bundle::Nexus({{"sales", sales}});
}

// A studio with one board holding one table over the sales data.
bi::Studio overview_studio(const pluss::bundle::Nexus& nexus) {
  bi::Studio studio("sales_studio");
  studio.source(nexus);
  studio.board("overview").title("Sales overview").grid(1, 1);
  studio.board("overview").view(bi::table("rows").query(bi::from("sales")).place(0, 0));
  return studio;
}

// The number of rows the overview table shows on a studio.
std::int64_t rows_shown(const bi::Studio& studio) {
  return studio.render().boards.front().tiles.front().table->height();
}

fs::path private_dir(const std::string& name) {
  const fs::path dir = fs::temp_directory_path() / name;
  fs::remove_all(dir);
  fs::create_directories(dir);
  return dir;
}

std::string read_file(const fs::path& path) {
  std::ifstream in(path, std::ios::binary);
  return std::string((std::istreambuf_iterator<char>(in)), std::istreambuf_iterator<char>());
}

template <class Fn>
bool throws(Fn fn) {
  try {
    fn();
  } catch (const std::exception&) {
    return true;
  }
  return false;
}

int main() {
  bi::Studio studio("Sales 2024");
  std::cout << "key: " << studio.key() << '\n';
  return 0;
}
```

Output

```text
key: Sales 2024
```

### Example 3

Example 3 shows that two studios with different keys render the same rows.

```cpp
#include <pluss/bi/bi.h>
#include <pluss/bi/serve.h>
#include <pluss/bundle/bundle.h>
#include <pluss/mic.h>

#include <cstdint>
#include <filesystem>
#include <fstream>
#include <iostream>
#include <iterator>

}

// A studio with one board holding one table over the sales data.

// The number of rows the overview table shows on a studio.

fs::path private_dir(const std::string& name) {

std::string read_file(const fs::path& path) {

}

template <class Fn>
bool throws(Fn fn) {
  try {
    fn();
  } catch (const std::exception&) {
    return true;
  }
  return false;
}

int main() {
  // The key names the studio. It does not change what the boards render.
  const auto nexus = make_nexus();
  bi::Studio first = overview_studio(nexus);
  bi::Studio second("another_name");
  second.source(nexus);
  second.board("overview").title("Sales overview").grid(1, 1);
  second.board("overview").view(bi::table("rows").query(bi::from("sales")).place(0, 0));
  std::cout << "same rows under either key: " << (rows_shown(first) == rows_shown(second)) << '\n';
  return 0;
}
```

Output

```text
same rows under either key: 1
```
