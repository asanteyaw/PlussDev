```cpp
  const Series y_reg = pluss::make_series(std::vector<double>{-1.0, -0.5, 0.0, 0.5, 1.0, 1.5}, {6}, f64());
  const LabeledBatch reg_train{
      .features = FeatureBatch{.x = x_reg, .layout = ObservationLayout::Rows},
      .y = y_reg,
  };

  std::cout << "Regression presets:\n";
  report_regressor("ridge", pluss::ai::lymph::quick::ridge_regressor(reg_train, 1e-8), reg_train);
  report_regressor("forest", pluss::ai::lymph::quick::forest_regressor(reg_train, 31, 0.9), reg_train);
  report_regressor("rbf_kernel", pluss::ai::lymph::quick::rbf_kernel_regressor(reg_train, 50.0, 0.7, 0.0), reg_train);
  report_regressor("gaussian_bayes", pluss::ai::lymph::quick::gaussian_bayes_regressor(reg_train, 1e-6), reg_train);

  std::cout << "lymph presets demo completed.\n";
  return 0;
}
```

### Example 2

*Source:* `PlussAI/examples/NodeJ/lymph/lymph_quick_regression_demo.cpp`

```cpp
  run_and_report("ridge", pluss::ai::lymph::quick::ridge_regressor(train, 1e-8), train);
  run_and_report("tree", pluss::ai::lymph::quick::tree_regressor(
                             train,
                             TreeFamilyOptions{
                                 .criterion = TreeCriterion::Mse,
                             }),
                 train);
  run_and_report("forest", pluss::ai::lymph::quick::forest_regressor(train, 31, 0.9), train);
  run_and_report("neighbor", pluss::ai::lymph::quick::neighbor_regressor(
                                 train,
                                 NeighborFamilyOptions{
                                     .k = 3,
                                     .distance_weighted = true,
                                     .standardize = true,
                                 }),
                 train);
  run_and_report("rbf_kernel", pluss::ai::lymph::quick::rbf_kernel_regressor(train, 50.0, 0.7, 0.0), train);
  run_and_report("gaussian_bayes", pluss::ai::lymph::quick::gaussian_bayes_regressor(train, 1e-6), train);

  std::cout << "lymph quick regression demo completed.\n";
  return 0;
}
```
