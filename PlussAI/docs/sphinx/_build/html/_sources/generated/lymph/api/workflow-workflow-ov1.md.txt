## Examples

### Example 1

*Source:* `PlussAI/examples/NodeJ/core/bike_sharing_end_to_end.cpp`

```cpp
    train_options.max_iterations = 2500;
    train_options.tolerance = 1e-8;
    train_options.random.seed = 7;
    train_options.random.deterministic = true;

    AssessOptions assess_options{};
    assess_options.metrics = {MetricKind::Rmse, MetricKind::R2, MetricKind::Mae};

    auto linear_workflow = lymph::Workflow(lymph::linear_factory::linear_regression(LinearFamilyOptions{
        .objective = LinearObjective::LeastSquares,
        .regularization = RegularizationKind::None,
        .regularization_strength = 0.0,
        .fit_intercept = true,
    }));
    linear_workflow
        .simple_imputer(lymph::prep::SimpleImputerOptions{.strategy = lymph::prep::SimpleImputerStrategy::Median})
        .standard_scaler();
    linear_workflow.learn_(train_x, train_y, train_options);
    const auto linear_metrics = linear_workflow.assess(test_x, test_y, assess_options);
    std::cout << "linear test metrics [rmse, r2, mae]: " << linear_metrics.rmse << " " << linear_metrics.r2 << " "
              << linear_metrics.mae << "\n";
    std::cout << "linear params: intercept=" << linear_workflow.intercept() << " coeff=" << linear_workflow.coeff() << "\n";

    const Series linear_pred_series = linear_workflow.infer(test_x);
    const fs::path linear_model_path = model_dir / "bike_sharing_linear.bin";
```

### Example 2

*Source:* `PlussAI/examples/NodeJ/core/nodej_workflow_tuning_demo.cpp`

```cpp
    cv_options.plan = CVPlanKind::KFold;
    cv_options.folds = 5;
    cv_options.shuffle = true;
    cv_options.random_state = 73;
    cv_options.deterministic = true;

    std::vector<nja::selection::Candidate> candidates{};
    {
      auto wf = nja::Workflow(nja::linear_factory::linear_regression(LinearFamilyOptions{
          .objective = LinearObjective::LeastSquares,
          .regularization = RegularizationKind::L2,
          .regularization_strength = 1e-6,
          .fit_intercept = true,
      }));
      wf.simple_imputer(nja::prep::SimpleImputerOptions{.strategy = nja::prep::SimpleImputerStrategy::Median})
          .standard_scaler();
      candidates.emplace_back("linear_ridge", std::move(wf));
    }
    {
      auto wf = nja::Workflow(nja::tree_factory::decision_tree_regressor(TreeFamilyOptions{
          .criterion = TreeCriterion::Mse,
          .max_depth = 18,
          .min_samples_leaf = 8,
          .min_samples_split = 24,
      }));
```
